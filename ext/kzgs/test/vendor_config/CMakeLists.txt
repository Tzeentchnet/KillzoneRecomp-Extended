cmake_minimum_required(VERSION 3.24)

if(KZGS_FAKE_COMPLETE_CASE OR KZGS_LINK_PROBE_CASE)
	project(KzgsVendorConfigProbe LANGUAGES CXX)
else()
	project(KzgsVendorConfigProbe LANGUAGES NONE)
	# Exercise the platform checks without requiring another installed compiler or SDK.
	set(WIN32 TRUE)
	set(MSVC TRUE)
	set(CMAKE_SYSTEM_NAME Windows)
	set(CMAKE_CXX_COMPILER_ID MSVC)
	set(CMAKE_CXX_COMPILER_ARCHITECTURE_ID x64)
	set(CMAKE_SIZEOF_VOID_P 8)
	if(KZGS_TEST_BAD_ARCH)
		set(CMAKE_CXX_COMPILER_ARCHITECTURE_ID ARM64)
	endif()
	if(KZGS_TEST_BAD_POINTER_SIZE)
		set(CMAKE_SIZEOF_VOID_P 4)
	endif()
	if(KZGS_TEST_BAD_TOOLCHAIN)
		set(MSVC FALSE)
		set(CMAKE_CXX_COMPILER_ID GNU)
	endif()
endif()

set(KZGS_PROVIDER_SOURCE_DIR "${KZGS_SOURCE}/ext/kzgs/providers")
set(KZGS_PROVIDER_PUBLIC_INCLUDE "${KZGS_SOURCE}/ext/kzgs/include")
set(KZGS_PROVIDER_VULKAN_INCLUDE "${KZGS_SOURCE}/ext/pcsx2/3rdparty/vulkan/include")
include("${KZGS_PROVIDER_SOURCE_DIR}/KzgsVendorProviders.cmake")
add_custom_target(kzgs_config_sentinel ALL COMMAND "${CMAKE_COMMAND}" -E echo "sentinel only")
kzgs_add_vendor_providers()

if(KZGS_FAKE_COMPLETE_CASE)
	foreach(vendor IN ITEMS DLSS XESS)
		string(TOLOWER "${vendor}" vendor_lower)
		set(target "kzgs_upscaler_${vendor_lower}")
		if(NOT KZGS_BUILD_${vendor}_PROVIDER)
			if(TARGET ${target})
				message(FATAL_ERROR "Disabled fake-SDK target ${target} leaked into the graph.")
			endif()
			continue()
		endif()
		if(NOT TARGET ${target})
			message(FATAL_ERROR "Enabled fake-SDK test expected target ${target}.")
		endif()
		get_target_property(excluded ${target} EXCLUDE_FROM_ALL)
		get_target_property(output_dir ${target} RUNTIME_OUTPUT_DIRECTORY)
		if(NOT excluded OR NOT output_dir MATCHES "local-sdk-providers")
			message(FATAL_ERROR "${target} leaks into the default build or ordinary output directory.")
		endif()
	endforeach()
else()
	if(TARGET kzgs_upscaler_dlss OR TARGET kzgs_upscaler_xess)
		message(FATAL_ERROR "Default-OFF SDK provider target leaked into the build.")
	endif()
endif()
